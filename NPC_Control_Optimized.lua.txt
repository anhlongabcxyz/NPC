local function isnpc(ins)
	local humanoid = ins:FindFirstChildOfClass("Humanoid")
	local player = Players:GetPlayerFromCharacter(ins)
	if humanoid and not player then
		return ins
	end
	return nil
end

local function partowner(part)
	return part.ReceiveAge == 0
end

-- UI Builder functions
local function createButton(titleText, desText, func)
	local sample = ButtonTemplate:Clone()
	sample.Parent = MainScrolling
	sample.title.Text = titleText or ""
	sample.des.Text = desText or ""

	sample.click.MouseButton1Click:Connect(function()
		if func then
			task.spawn(function()
				TweenService:Create(sample.UIStroke, TweenInfo.new(0.1), {Thickness = 1.5}):Play()
				task.wait(.2)
				TweenService:Create(sample.UIStroke, TweenInfo.new(0.1), {Thickness = 1}):Play()
			end)
			local _, err = pcall(func)
			if err then warn(err) end
		end
	end)
end

local function createToggle(titleText, desText, func)
	local sample = ToggleTemplate:Clone()
	sample.Parent = MainScrolling
	sample.title.Text = titleText or ""
	sample.des.Text = desText or ""

	local toggled = false
	sample.click.MouseButton1Click:Connect(function()
		toggled = not toggled
		if func then
			task.spawn(function()
				TweenService:Create(sample.toggle, TweenInfo.new(.2), {BackgroundColor3 = toggled and Color3.fromRGB(235, 235, 235) or Color3.fromRGB(85, 85, 85)}):Play()
				TweenService:Create(sample.UIStroke, TweenInfo.new(.1), {Thickness = 1.5}):Play()
				task.wait(.2)
				TweenService:Create(sample.UIStroke, TweenInfo.new(.1), {Thickness = 1}):Play()
			end)
			local _, err = pcall(func, toggled)
			if err then warn(err) end
		end
	end)

	local addition = {}
	function addition:swich(bool)
		toggled = bool
		TweenService:Create(sample.toggle, TweenInfo.new(.2), {BackgroundColor3 = toggled and Color3.fromRGB(235, 235, 235) or Color3.fromRGB(85, 85, 85)}):Play()
	end
	return addition
end

-- NPC registry: scan existing NPCs once, then register/remove them incrementally.
-- This avoids querying every BasePart on every frame when the map contains many NPCs.
local NPC_AURA_RADIUS = 30
local NPC_AURA_INTERVAL = 0.12
local npcRegistry = {}
local npcLastAction = {}

local function getNPCModel(instance)
	if not instance then return nil end
	if instance:IsA("Model") then
		return instance
	end
	return instance:FindFirstAncestorOfClass("Model")
end

local function registerNPC(instance)
	local model = getNPCModel(instance)
	if model and model:IsDescendantOf(Workspace) and isnpc(model) then
		npcRegistry[model] = true
	end
end

-- One initial registration pass.
for _, instance in ipairs(Workspace:GetDescendants()) do
	if instance:IsA("Model") then
		registerNPC(instance)
	elseif instance:IsA("Humanoid") then
		registerNPC(instance)
	end
end

-- Future NPCs are detected as they enter the Workspace; no full rescans needed.
local npcAddedConnection = Workspace.DescendantAdded:Connect(function(instance)
	if instance:IsA("Model") or instance:IsA("Humanoid") then
		registerNPC(instance)
	end
end)

local npcRemovingConnection = Workspace.DescendantRemoving:Connect(function(instance)
	if instance:IsA("Model") then
		npcRegistry[instance] = nil
		npcLastAction[instance] = nil
	elseif instance:IsA("Humanoid") then
		local model = instance.Parent
		if model and model:IsA("Model") then
			task.defer(function()
				if model.Parent and not model:FindFirstChildOfClass("Humanoid") then
					npcRegistry[model] = nil
					npcLastAction[model] = nil
				end
			end)
		end
	end
end)

local con1
createToggle("kill aura any npc", "NPC detection • 30 studs • optimized scan", function(a)
	if a then
		local elapsed = 0
		con1 = RunService.Heartbeat:Connect(function(dt)
			elapsed += dt
			if elapsed < NPC_AURA_INTERVAL then return end
			elapsed = 0

			local character = lp.Character
			local playerRoot = character and character:FindFirstChild("HumanoidRootPart")
			if not playerRoot then return end

			local origin = playerRoot.Position
			for model in pairs(npcRegistry) do
				if not model.Parent or not model:IsDescendantOf(Workspace) then
					npcRegistry[model] = nil
					npcLastAction[model] = nil
					continue
				end

				local humanoid = model:FindFirstChildOfClass("Humanoid")
				local root = model:FindFirstChild("HumanoidRootPart")
				if humanoid and humanoid.Health > 0 and root and not root.Anchored
					and model ~= character and (root.Position - origin).Magnitude <= NPC_AURA_RADIUS then
					local now = os.clock()
					if now - (npcLastAction[model] or 0) >= 0.25 then
						npcLastAction[model] = now
						if partowner(root) then
							humanoid:ChangeState(save["9"].val)
						end
					end
				end
			end
		end)
	else
		if con1 then con1:Disconnect(); con1 = nil end
	end
end)

-- Minimize Logic
local minimizebool = false
local function toggleMinimize()
	minimizebool = not minimizebool
	if minimizebool then
		IconFrame.Visible = true
		IconScale.Scale = 0
		TweenService:Create(UIScale, fast, {Scale = 0}):Play()
		TweenService:Create(IconScale, fast, {Scale = 1}):Play()
		task.wait(.5)
		Frame.Visible = false
	else
		Frame.Visible = true
		TweenService:Create(UIScale, fast, {Scale = 1}):Play()
		TweenService:Create(IconScale, fast, {Scale = 0}):Play()
		task.wait(.5)
		IconFrame.Visible = false
	end
end

MinimizeBtn.MouseButton1Click:Connect(toggleMinimize)
IconText.MouseButton1Click:Connect(toggleMinimize)

-- Intro Animation
local function playIntro()
	DarkOverlay.Visible = true
	DarkOverlay.BackgroundTransparency = 1
	IntroIcon.TextTransparency = 1
	IntroTitle.TextTransparency = 1
	IntroSub.TextTransparency = 1
	IntroScale.Scale = 0
	IntroFrame.BackgroundTransparency = 1
	IntroFrame.Visible = true
	UIScale.Scale = 0

	local blur = Instance.new("BlurEffect")
	blur.Name = "intro_blur"
	blur.Size = 0
	blur.Parent = Lighting

	TweenService:Create(blur, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = 8}):Play()
	TweenService:Create(DarkOverlay, TweenInfo.new(0.35, Enum.EasingStyle.Quad), {BackgroundTransparency = 0.38}):Play()
	task.wait(.2)
	TweenService:Create(IntroFrame, medium, {BackgroundTransparency = 0}):Play()
	TweenService:Create(IntroScale, medium, {Scale = 1}):Play()
	task.wait(.2)
	TweenService:Create(IntroIcon, medium, {TextTransparency = 0}):Play()
	task.wait(.3)
	TweenService:Create(IntroTitle, medium, {TextTransparency = 0}):Play()
	TweenService:Create(IntroSub, medium, {TextTransparency = 0}):Play()
	task.wait(2.4)
	TweenService:Create(blur, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {Size = 0}):Play()
	TweenService:Create(IntroFrame, medium, {BackgroundTransparency = 1}):Play()
	TweenService:Create(IntroScale, medium, {Scale = 0}):Play()
	task.wait(.3)
	TweenService:Create(DarkOverlay, slow, {BackgroundTransparency = 1}):Play()
	task.wait(.5)
	TweenService:Create(UIScale, medium, {Scale = 1}):Play()
	DarkOverlay.Visible = false
	IntroFrame.Visible = false
	blur:Destroy()
end

playIntro()
